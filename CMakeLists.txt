cmake_minimum_required(VERSION 3.16)
file(STRINGS "${CMAKE_CURRENT_LIST_DIR}/VERSION" PROJECT_VER LIMIT_COUNT 1)
if(NOT PROJECT_VER MATCHES "^[0-9]+\\.[0-9]+\\.[0-9]+(-[A-Za-z0-9.]+)?$")
  message(FATAL_ERROR "Invalid project version")
endif()
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_LIST_DIR}/VERSION")
option(ZAM_EXPERIMENTAL_LUA "Enable Lua and Program Manager" ON)
# Record explicitly requested values before Kconfig can silently clamp a range
# or discard an option whose dependencies are not met. Existing sdkconfig
# takes precedence over the defaults, as in ESP-IDF.
set(_zam_requested_psram "")
set(_zam_requested_octal "")
set(_zam_requested_limit "")
if(DEFINED SDKCONFIG_DEFAULTS AND NOT SDKCONFIG_DEFAULTS STREQUAL "")
  set(_zam_inputs ${SDKCONFIG_DEFAULTS})
else()
  set(_zam_inputs "${CMAKE_CURRENT_LIST_DIR}/sdkconfig.defaults")
endif()
list(APPEND _zam_inputs "__ZAM_DEFAULTS_END__")
if(DEFINED SDKCONFIG)
  list(APPEND _zam_inputs "${SDKCONFIG}")
else()
  list(APPEND _zam_inputs "${CMAKE_CURRENT_LIST_DIR}/sdkconfig")
endif()
foreach(_zam_file IN LISTS _zam_inputs)
  if(_zam_file STREQUAL "__ZAM_DEFAULTS_END__")
    set(_zam_defaults_psram "${_zam_requested_psram}")
    set(_zam_defaults_octal "${_zam_requested_octal}")
    set(_zam_defaults_limit "${_zam_requested_limit}")
    continue()
  endif()
  if(EXISTS "${_zam_file}")
    file(STRINGS "${_zam_file}" _zam_lines REGEX "^(CONFIG_ZAM_LUA_PSRAM_EXPERIMENT=|# CONFIG_ZAM_LUA_PSRAM_EXPERIMENT is not set|CONFIG_ZAM_LUA_HEAP_LIMIT=|CONFIG_SPIRAM_MODE_OCT=|# CONFIG_SPIRAM_MODE_OCT is not set)")
    foreach(_zam_line IN LISTS _zam_lines)
      if(_zam_line MATCHES "^CONFIG_ZAM_LUA_PSRAM_EXPERIMENT=(.*)$")
        set(_zam_requested_psram "${CMAKE_MATCH_1}")
      elseif(_zam_line MATCHES "^# CONFIG_ZAM_LUA_PSRAM_EXPERIMENT is not set")
        set(_zam_requested_psram "n")
      elseif(_zam_line MATCHES "^CONFIG_ZAM_LUA_HEAP_LIMIT=([0-9]+)$")
        set(_zam_requested_limit "${CMAKE_MATCH_1}")
      elseif(_zam_line MATCHES "^CONFIG_SPIRAM_MODE_OCT=(.*)$")
        set(_zam_requested_octal "${CMAKE_MATCH_1}")
      elseif(_zam_line MATCHES "^# CONFIG_SPIRAM_MODE_OCT is not set")
        set(_zam_requested_octal "n")
      endif()
    endforeach()
  endif()
endforeach()
include($ENV{IDF_PATH}/tools/cmake/project.cmake)
idf_build_set_property(DEPENDENCIES_LOCK dependencies.lock.${IDF_TARGET})
project(zamprolab)
if(_zam_defaults_psram STREQUAL "y" AND NOT CONFIG_ZAM_LUA_PSRAM_EXPERIMENT)
  message(FATAL_ERROR "Requested Lua PSRAM allocator requires the verified ESP32-S3 RGB profile and explicit-capability PSRAM")
endif()
if(NOT _zam_defaults_limit STREQUAL "" AND NOT _zam_defaults_limit STREQUAL "${CONFIG_ZAM_LUA_HEAP_LIMIT}")
  message(FATAL_ERROR "Requested Lua heap limit ${_zam_defaults_limit} is incompatible with this profile or sdkconfig (resolved: ${CONFIG_ZAM_LUA_HEAP_LIMIT})")
endif()
if(_zam_defaults_octal STREQUAL "y" AND (NOT CONFIG_IDF_TARGET_ESP32S3 OR NOT CONFIG_ZAM_RGB_VERIFIED_S3 OR NOT CONFIG_SPIRAM OR NOT CONFIG_SPIRAM_MODE_OCT))
  message(FATAL_ERROR "Requested octal PSRAM requires the verified ESP32-S3 RGB profile and enabled octal PSRAM")
endif()
if(_zam_requested_psram STREQUAL "y" AND NOT CONFIG_ZAM_LUA_PSRAM_EXPERIMENT)
  message(FATAL_ERROR "Requested Lua PSRAM allocator requires the verified ESP32-S3 RGB profile and explicit-capability PSRAM")
endif()
if(NOT _zam_requested_limit STREQUAL "" AND NOT _zam_requested_limit STREQUAL "${CONFIG_ZAM_LUA_HEAP_LIMIT}")
  message(FATAL_ERROR "Requested Lua heap limit ${_zam_requested_limit} is incompatible with this profile (resolved: ${CONFIG_ZAM_LUA_HEAP_LIMIT})")
endif()
if(_zam_requested_octal STREQUAL "y" AND (NOT CONFIG_IDF_TARGET_ESP32S3 OR NOT CONFIG_ZAM_RGB_VERIFIED_S3 OR NOT CONFIG_SPIRAM OR NOT CONFIG_SPIRAM_MODE_OCT))
  message(FATAL_ERROR "Requested octal PSRAM requires the verified ESP32-S3 RGB profile and enabled octal PSRAM")
endif()
if(CONFIG_ZAM_LUA_PSRAM_EXPERIMENT AND (NOT CONFIG_ESPTOOLPY_FLASHSIZE_16MB OR NOT CONFIG_SPIRAM_MODE_OCT OR NOT CONFIG_SPIRAM_SPEED_80M OR NOT CONFIG_SPIRAM_USE_CAPS_ALLOC OR NOT CONFIG_SPIRAM_BOOT_INIT OR CONFIG_SPIRAM_IGNORE_NOTFOUND))
  message(FATAL_ERROR "Lua PSRAM experiment requires 16 MB flash and initialized octal PSRAM at 80 MHz with explicit-capability allocation and no ignore-not-found")
endif()
