set(runtime_sources)
if(ZAM_EXPERIMENTAL_LUA)
  if(NOT DEFINED ENV{LUA_DIR} OR NOT EXISTS "$ENV{LUA_DIR}/src/lua.h")
    message(FATAL_ERROR "Set LUA_DIR to verified Lua 5.5.1 source")
  endif()
  include("${CMAKE_CURRENT_LIST_DIR}/lua-5.5.1-sha256.cmake")
  set(lua_sources)
  list(LENGTH ZAM_LUA_FILES manifest_length)
  math(EXPR last "${manifest_length} - 1")
  foreach(index RANGE 0 ${last} 2)
    math(EXPR hash_index "${index} + 1")
    list(GET ZAM_LUA_FILES ${index} name)
    list(GET ZAM_LUA_FILES ${hash_index} expected)
    set(path "$ENV{LUA_DIR}/src/${name}")
    if(NOT EXISTS "${path}")
      message(FATAL_ERROR "Missing Lua source ${name}")
    endif()
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${path}")
    file(SHA256 "${path}" actual)
    if(NOT actual STREQUAL expected)
      message(FATAL_ERROR "Lua 5.5.1 integrity check failed for ${name}")
    endif()
    if(name MATCHES "\\.c$" AND NOT name MATCHES "^(lua|luac|onelua)\\.c$")
      list(APPEND lua_sources "${path}")
    endif()
  endforeach()
  list(APPEND runtime_sources zam_runtime_lua.c zam_lua_allocator.c zam_lua_bindings.c zam_lua_modules.c zam_lua_gpio.c zam_lua_i2c.c ${lua_sources})
endif()
idf_component_register(SRCS ${runtime_sources} INCLUDE_DIRS "include" PRIV_INCLUDE_DIRS "." REQUIRES zam_files esp_timer heap zam_rgb PRIV_REQUIRES esp_driver_gpio esp_driver_i2c)
if(ZAM_EXPERIMENTAL_LUA)
  target_include_directories(${COMPONENT_LIB} PRIVATE "$ENV{LUA_DIR}/src")
endif()
